\begin{abstract}
\noindent
This document provides a comprehensive technical specification for the \texttt{pygnss\_rt} Python package, a production-grade framework for Real-Time Precise Point Positioning with Ambiguity Resolution (PPP-AR). The system integrates with the Bernese GNSS Software v5.4 for core geodetic computations while providing a sophisticated Python orchestration layer for data management, product acquisition, and operational automation.

\vspace{0.5cm}
\noindent
\textbf{Key Features:}
\begin{itemize}[noitemsep,topsep=0pt]
    \item Multi-GNSS support: GPS (G) + Galileo (E) combination for optimal AR
    \item VMF3 Vienna Mapping Functions for tropospheric modeling
    \item CODE OSB (Observable-Specific Signal Biases) for phase bias consistency
    \item LAMBDA-based integer ambiguity resolution
    \item Automated product downloads from CDDIS, CODE, IGN
    \item ZTD to IWV conversion for meteorological applications
    \item Five international station networks (IGS, EUREF, GB, RGP, Supersites)
\end{itemize}

\vspace{0.5cm}
\noindent
\textbf{GitHub Repository:} \url{https://github.com/georgisalene/pygnss_rt}

\vspace{0.5cm}
\noindent
This document includes the complete source code with colorful syntax highlighting, mathematical formulations, and architectural diagrams for all major system components.
\end{abstract}

@cli.command()
@click.argument("network", type=click.Choice(["IG", "EU", "GB", "RG", "SS"]))
@click.option("--start-date", "-s", type=str, required=True,
              help="Start date (YYYY-MM-DD)")
@click.option("--end-date", "-e", type=str, required=True,
              help="End date (YYYY-MM-DD)")
@click.option("--stations", "-S", type=str, help="Comma-separated stations")
@click.option("--exclude", "-x", type=str, help="Stations to exclude")
@click.option("--skip-products", is_flag=True, help="Skip product download")
@click.option("--skip-data", is_flag=True, help="Skip station data download")
@click.option("--skip-dcm", is_flag=True, help="Skip DCM archiving")
@click.option("--dry-run", is_flag=True, help="Show what would be done")
@click.pass_context
def daily_ppp(
    ctx: click.Context,
    network: str,
    start_date: str,
    end_date: str,
    stations: str | None,
    exclude: str | None,
    skip_products: bool,
    skip_data: bool,
    skip_dcm: bool,
    dry_run: bool,
) -> None:
    """Run daily PPP-AR processing for a network.

    NETWORK: Network identifier (IG, EU, GB, RG, SS)

    Examples:
        pygnss_rt daily-ppp IG -s 2025-12-23 -e 2025-12-23
        pygnss_rt daily-ppp EU -s 2025-01-01 -e 2025-01-31
    """
    from pygnss_rt.processing.daily_ppp import DailyPPPProcessor, DailyPPPArgs
    from pygnss_rt.utils.dates import GNSSDate

    # Parse dates
    start = GNSSDate.from_string(start_date)
    end = GNSSDate.from_string(end_date)

    # Build arguments
    args = DailyPPPArgs(
        network_id=network,
        start_date=start,
        end_date=end,
        stations=stations.split(",") if stations else [],
        exclude_stations=exclude.split(",") if exclude else [],
        skip_products=skip_products,
        skip_data=skip_data,
        skip_dcm=skip_dcm,
        dry_run=dry_run,
        verbose=ctx.obj.get("verbose", False),
    )

    # Run processor
    processor = DailyPPPProcessor(args)
    results = processor.process()

    # Print summary
    success_count = sum(1 for r in results if r.success)
    print(f"\n{'='*50}")
    print(f"PROCESSING SUMMARY")
    print(f"{'='*50}")
    print(f"Total: {success_count}/{len(results)} days successful")
\end{lstlisting}

class DailyPPPResult:

\section{Main Processor Class}

\begin{lstlisting}[caption={DailyPPPProcessor Implementation}, label={lst:processor}]
class DailyPPPProcessor:
    """Daily PPP processor managing complete workflow."""

    def __init__(self, args: DailyPPPArgs):
        """Initialize processor with arguments."""
        self.args = args
        self.paths = get_paths()

        # Get network profile
        network_id = NetworkID(args.network_id) if isinstance(
            args.network_id, str) else args.network_id
        self.profile = get_network_profile(network_id)

        # Initialize components
        self.bpe_runner: BPERunner | None = None
        self.downloader: ProductDownloader | None = None

    def process(self) -> list[DailyPPPResult]:
        """Run daily PPP processing for date range.

        Returns:
            List of results, one per day processed
        """
        results: list[DailyPPPResult] = []

        if not self.args.start_date or not self.args.end_date:
            raise ValueError("Start and end dates are required")

        current = self.args.start_date
        while current.mjd <= self.args.end_date.mjd:
            result = self._process_single_day(current)
            results.append(result)
            current = current.add_days(1)

        return results

    def _process_single_day(self, date: GNSSDate) -> DailyPPPResult:
        """Process a single day."""
        result = DailyPPPResult(
            network_id=self.profile.network_id.value,
            session_name=self._get_session_name(date),
            date=date,
            success=False,
        )

        try:
            # Step 1: Download products
            if not self.args.skip_products:
                self._check_products(date)

            # Step 2: Download station data
            if not self.args.skip_data:
                stations = self._download_station_data(date)
                result.stations_total = len(stations)

            # Step 3: Setup campaign
            campaign_root = self._setup_campaign(date)

            # Step 4: Run BSW processing
            bpe_result = self._run_bsw_processing(date, campaign_root)

            if bpe_result.success:
                result.success = True
                result.stations_processed = bpe_result.sessions_finished

                # Step 5: DCM archiving
                if not self.args.skip_dcm:
                    self._run_dcm(date, campaign_root)

        except Exception as e:
            result.error_message = str(e)

        result.end_time = datetime.now(timezone.utc)
        return result

\section{INP File Customization}

\begin{lstlisting}[caption={INP File Customization Methods}, label={lst:inpcustom}]
def put_key(
    self,
    inp_file: Path,
    key: str,
    value: str,
) -> bool:
    """Set a key-value pair in an INP file.

    Mimics the Perl putKey function from BSW interface.

    Args:
        inp_file: Path to INP file
        key: Key name (e.g., "USE_G", "TROPOS")
        value: Value to set

    Returns:
        True if key was found and updated
    """
    if not inp_file.exists():
        logger.warning(f"INP file not found: {inp_file}")
        return False

    content = inp_file.read_text()

    # Pattern: KEY  "value" or KEY  'value' or KEY  value
    pattern = rf'^(\s*{re.escape(key)}\s+)(["\']?)([^"\']*)\2\s*$'

    def replacer(match):
        prefix = match.group(1)
        quote = match.group(2) or "'"
        return f"{prefix}{quote}{value}{quote}"

    new_content, count = re.subn(pattern, replacer, content, flags=re.MULTILINE)

    if count > 0:
        inp_file.write_text(new_content)
        logger.debug(f"Set {key}={value} in {inp_file.name}")
        return True

    logger.warning(f"Key {key} not found in {inp_file}")
    return False


def customize_inp_files(
    self,
    temp_opt_dir: Path,
    bsw_options: dict[str, dict[str, dict[str, str]]],
    variable_subs: dict[str, str] | None = None,
) -> int:
    """Customize INP files based on BSW options from YAML/XML.

    Args:
        temp_opt_dir: Temp OPT directory
        bsw_options: Nested dict {opt_dir: {inp_file: {key: value}}}
        variable_subs: Variable substitutions for opt_* values

    Returns:
        Number of keys set
    """
    keys_set = 0

    for opt_name, inp_files in bsw_options.items():
        opt_path = temp_opt_dir / opt_name

        if not opt_path.exists():
            logger.warning(f"OPT directory not found: {opt_path}")
            continue

        for inp_name, keys in inp_files.items():
            inp_file = opt_path / f"{inp_name}.INP"

            if not inp_file.exists():
                logger.warning(f"INP file not found: {inp_file}")
                continue

            for key, value in keys.items():
                # Apply variable substitutions
                if variable_subs and value.startswith("opt_"):
                    value = variable_subs.get(value, value)

                if self.put_key(inp_file, key, value):
                    keys_set += 1

    return keys_set

    def from_mjd(cls, mjd: float) -> GNSSDate:
        """Create from Modified Julian Date."""
        dt = date_from_mjd(mjd)
        return cls(
            year=dt.year, month=dt.month, day=dt.day,
            hour=dt.hour, minute=dt.minute, second=dt.second
        )

    @classmethod
    def from_gps_week(cls, week: int, dow: int, sow: float = 0.0) -> GNSSDate:
        """Create from GPS week and day of week."""
        mjd = GPS_EPOCH_MJD + week * 7 + dow + sow / 86400.0
        return cls.from_mjd(mjd)

    @classmethod
    def from_doy(cls, year: int, doy: int, hour: int = 0) -> GNSSDate:
        """Create from year and day of year."""
        dt = datetime(year, 1, 1) + timedelta(days=doy - 1, hours=hour)
        return cls(year=dt.year, month=dt.month, day=dt.day, hour=hour)

    @classmethod
    def from_string(cls, date_str: str) -> GNSSDate:
        """Parse date from string (YYYY-MM-DD or YYYY/DOY)."""
        if "/" in date_str and len(date_str.split("/")[1]) == 3:
            # YYYY/DOY format
            year, doy = date_str.split("/")
            return cls.from_doy(int(year), int(doy))
        else:
            # YYYY-MM-DD format
            dt = datetime.strptime(date_str, "%Y-%m-%d")
            return cls(year=dt.year, month=dt.month, day=dt.day)

    @classmethod
    def now(cls) -> GNSSDate:
        """Get current UTC time."""
        dt = datetime.now(timezone.utc)
        return cls(
            year=dt.year, month=dt.month, day=dt.day,
            hour=dt.hour, minute=dt.minute, second=dt.second
        )

    def add_days(self, days: int) -> GNSSDate:
        """Add days to date."""
        return GNSSDate.from_mjd(self.mjd + days)

    def add_hours(self, hours: int) -> GNSSDate:
        """Add hours to date."""
        return GNSSDate.from_mjd(self.mjd + hours / 24.0)
\end{lstlisting}

def mjd_from_date(
    year: int, month: int, day: int,
    hour: int = 0, minute: int = 0, second: float = 0.0,
) -> float:
    """Calculate Modified Julian Date from calendar date.

    Uses the standard astronomical algorithm.
    """
    a = (14 - month) // 12
    y = year + 4800 - a
    m = month + 12 * a - 3

    # Julian Day Number
    jdn = day + (153 * m + 2) // 5 + 365 * y + y // 4 - y // 100 + y // 400 - 32045

    # Add fractional day
    frac = (hour + minute / 60.0 + second / 3600.0) / 24.0

    return jdn - MJD_OFFSET + frac


def gps_week_from_mjd(mjd: float) -> tuple[int, int]:
    """Get GPS week and day of week from MJD.

    Returns:
        Tuple of (gps_week, day_of_week)
    """
    days_since_epoch = mjd - GPS_EPOCH_MJD
    gps_week = int(days_since_epoch / 7)
    dow = int(days_since_epoch % 7)
    return gps_week, dow


def hour_to_alpha(hour: int) -> str:
    """Convert hour (0-23) to alpha character (a-x)."""
    if not 0 <= hour <= 23:
        raise ValueError(f"Hour must be 0-23, got {hour}")
    return chr(ord('a') + hour)


def alpha_to_hour(alpha: str) -> int:
    """Convert alpha character (a-x) to hour (0-23)."""
    alpha = alpha.lower()
    if not 'a' <= alpha <= 'x':
        raise ValueError(f"Alpha must be a-x, got {alpha}")
    return ord(alpha) - ord('a')

\chapter{ZTD to IWV Conversion}

\section{Physical Background}

The ZTD to IWV conversion implements the Bevis (1992, 1994) formulation for deriving Integrated Water Vapor from GNSS-estimated Zenith Tropospheric Delay.

\subsection{Mathematical Formulation}

The Zenith Total Delay (ZTD) is decomposed into hydrostatic and wet components:
\begin{equation}
\ztd = \zhd + \zwd
\end{equation}

The Zenith Hydrostatic Delay is computed using Saastamoinen (1972):
\begin{equation}
\zhd = \frac{0.0022768 \cdot P_s}{1 - 0.00266 \cos(2\phi) - 0.00028 H}
\end{equation}

The IWV is then derived from ZWD using:
\begin{equation}
\iwv = \frac{10^6 \cdot \zwd}{\rho_w \cdot R_v \cdot (k_2' + k_3/T_m)}
\end{equation}

where $T_m$ is the mean atmospheric temperature estimated from surface temperature.

\section{Implementation}

\begin{lstlisting}[caption={ZTD2IWV Core Implementation}, label={lst:ztd2iwv}]
"""
ZTD to IWV conversion module.

Converts Zenith Tropospheric Delay (ZTD) to Integrated Water Vapor (IWV).
Implements Bevis (1992, 1994) formulation with meteorological lookup.
"""

from __future__ import annotations

import math
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path

from pygnss_rt.utils.logging import get_logger


logger = get_logger(__name__)


# Physical constants (Bevis, 1994)
R_DRY = 287.0586      # Gas constant for dry air (J/kg/K)
R_VAPOR = 461.525     # Gas constant for water vapor (J/kg/K)
K1 = 77.6             # K/hPa
K2 = 70.4             # K/hPa
K3 = 373900           # K^2/hPa
G = 9.80665           # Standard gravity (m/s^2)

# ICAO standard atmosphere
ALPHA = -0.0065       # Temperature lapse rate (K/m)


@dataclass
class IWVResult:
    """IWV calculation result."""

    station_id: str
    timestamp: datetime
    mjd: float
    ztd: float           # Zenith Total Delay (mm)
    ztd_sigma: float     # ZTD uncertainty (mm)
    zhd: float           # Zenith Hydrostatic Delay (mm)
    zwd: float           # Zenith Wet Delay (mm)
    iwv: float           # Integrated Water Vapor (kg/m^2)
    iwv_sigma: float     # IWV uncertainty (kg/m^2)
    latitude: float
    longitude: float
    height: float        # Ellipsoidal height (m)
    pressure: float | None = None
    temperature: float | None = None
    met_station_id: str | None = None


class ZTD2IWV:
    """ZTD to IWV converter.

    Implements:
    - Saastamoinen ZHD model
    - Bevis mean temperature formulation
    - ICAO pressure extrapolation
    """

    def __init__(
        self,
        met_station_db: MeteoStationDatabase | None = None,
        max_met_distance: float = 100000.0,  # 100 km
    ):
        """Initialize converter."""
        self.met_db = met_station_db
        self.max_met_distance = max_met_distance

    def calculate_zhd_saastamoinen(
        self,
        pressure: float,
        latitude: float,
        height: float,
    ) -> float:
        """Calculate Zenith Hydrostatic Delay using Saastamoinen model.

        Args:
            pressure: Surface pressure (hPa)
            latitude: Station latitude (degrees)
            height: Ellipsoidal height (m)

        Returns:
            ZHD in millimeters
        """
        lat_rad = math.radians(latitude)

        # Saastamoinen formula
        zhd = (0.0022768 * pressure) / (
            1 - 0.00266 * math.cos(2 * lat_rad) - 0.00028 * height / 1000.0
        )

        return zhd * 1000.0  # Convert to mm

    def calculate_mean_temperature(
        self,
        surface_temp: float,
        latitude: float,
    ) -> float:
        """Estimate mean atmospheric temperature.

        Uses Bevis (1992) empirical relationship:
        Tm = 70.2 + 0.72 * Ts

        Args:
            surface_temp: Surface temperature (K)
            latitude: Station latitude (degrees)

        Returns:
            Mean temperature (K)
        """
        # Bevis empirical formula
        tm = 70.2 + 0.72 * surface_temp
        return tm

    def calculate_iwv_from_zwd(
        self,
        zwd: float,
        mean_temp: float,
    ) -> float:
        """Calculate IWV from Zenith Wet Delay.

        Args:
            zwd: Zenith Wet Delay (mm)
            mean_temp: Mean atmospheric temperature (K)

        Returns:
            IWV (kg/m^2)
        """
        # Conversion factor (Bevis, 1994)
        k2_prime = K2 - K1 * R_DRY / R_VAPOR

        # Pi factor
        pi_factor = 1e6 / (R_VAPOR * (k2_prime + K3 / mean_temp))

        # IWV = ZWD * Pi
        iwv = zwd * pi_factor / 1000.0  # zwd in mm, iwv in kg/m^2

        return iwv

    def process(
        self,
        station_id: str,
        ztd: float,
        ztd_sigma: float,
        timestamp: datetime,
        latitude: float,
        longitude: float,
        height: float,
        met_obs: MeteoObservation | None = None,
    ) -> IWVResult:
        """Convert single ZTD observation to IWV.

        Args:
            station_id: 4-char station ID
            ztd: Zenith Total Delay (mm)
            ztd_sigma: ZTD uncertainty (mm)
            timestamp: Observation time
            latitude: Station latitude (degrees)
            longitude: Station longitude (degrees)
            height: Ellipsoidal height (m)
            met_obs: Optional meteorological observation

        Returns:
            IWVResult with all derived quantities
        """
        # Get meteorological data
        if met_obs is None and self.met_db:
            met_obs = self._find_nearest_met_obs(latitude, longitude, timestamp)

        if met_obs is None:
            raise ValueError("No meteorological data available")

        # Extrapolate pressure to GPS height
        pressure = self._extrapolate_pressure(
            met_obs.msl_pressure,
            met_obs.height,
            height,
            met_obs.temperature,
        )

        # Calculate ZHD
        zhd = self.calculate_zhd_saastamoinen(pressure, latitude, height)

        # Calculate ZWD
        zwd = ztd - zhd

        # Calculate mean temperature
        tm = self.calculate_mean_temperature(met_obs.temperature, latitude)

        # Calculate IWV
        iwv = self.calculate_iwv_from_zwd(zwd, tm)

        # Propagate uncertainty
        iwv_sigma = self._propagate_uncertainty(ztd_sigma, zwd, tm)

        return IWVResult(
            station_id=station_id,
            timestamp=timestamp,
            mjd=mjd_from_datetime(timestamp),
            ztd=ztd,
            ztd_sigma=ztd_sigma,
            zhd=zhd,
            zwd=zwd,
            iwv=iwv,
            iwv_sigma=iwv_sigma,
            latitude=latitude,
            longitude=longitude,
            height=height,
            pressure=pressure,
            temperature=met_obs.temperature,
            met_station_id=met_obs.station_id,
        )
\end{lstlisting}

        Args:
            filepath: Path to XML or YAML file

        Returns:
            Number of stations loaded
        """
        if filepath.suffix in (".yaml", ".yml"):
            return self.load_yaml(filepath)
        elif filepath.suffix == ".xml":
            return self.load_xml(filepath)
        else:
            raise ValueError(f"Unknown file format: {filepath.suffix}")

    def load_yaml(self, filepath: Path) -> int:

        with open(filepath) as f:
            data = yaml.safe_load(f)

        stations = data.get("stations", [])
        for sta_data in stations:
            station = Station.from_yaml_dict(sta_data)
            self._stations[station.station_id] = station

        logger.info(f"Loaded {len(stations)} stations from {filepath.name}")
        return len(stations)

        """Get station by ID."""
        return self._stations.get(station_id.lower())

        active: bool | None = None,

        if network:
            stations = [s for s in stations if s.network == network]
        if use_nrt is not None:
            stations = [s for s in stations if s.use_nrt == use_nrt]
        if active is not None:
            stations = [s for s in stations if s.active == active]

        return stations

\chapter{Mathematical Foundations}

\section{PPP-AR Observation Model}

The ionosphere-free linear combination for dual-frequency observations:

\begin{equation}
L_{IF} = \frac{f_1^2 L_1 - f_2^2 L_2}{f_1^2 - f_2^2}
\end{equation}

\begin{equation}
P_{IF} = \frac{f_1^2 P_1 - f_2^2 P_2}{f_1^2 - f_2^2}
\end{equation}

\section{Kalman Filter State Vector}

The PPP-AR state vector contains:

\begin{equation}
\vecbold{x} = \begin{bmatrix}
\vecbold{r} \\
\delta t_r \\
\ztd \\
G_N, G_E \\
\vecbold{N}
\end{bmatrix}
\end{equation}

where:
\begin{itemize}
    \item $\vecbold{r}$: Position coordinates (X, Y, Z)
    \item $\delta t_r$: Receiver clock offset
    \item $\ztd$: Zenith Tropospheric Delay
    \item $G_N, G_E$: Tropospheric gradients (North, East)
    \item $\vecbold{N}$: Carrier phase ambiguities
\end{itemize}

\section{LAMBDA Ambiguity Resolution}

The integer ambiguity search minimizes:

\begin{equation}
\|\hat{\vecbold{a}} - \vecbold{a}\|_{\mat{Q}_{\hat{a}}}^2 = (\hat{\vecbold{a}} - \vecbold{a})^T \mat{Q}_{\hat{a}}^{-1} (\hat{\vecbold{a}} - \vecbold{a})
\end{equation}

The ratio test validates the solution:

\begin{equation}
R = \frac{\|\hat{\vecbold{a}} - \vecbold{a}_2\|^2}{\|\hat{\vecbold{a}} - \vecbold{a}_1\|^2} > R_{threshold}
\end{equation}

\section{VMF3 Mapping Functions}

The Vienna Mapping Functions VMF3 compute slant delays:

\begin{equation}
m_h(e) = \frac{1 + \frac{a_h}{1 + \frac{b_h}{1 + c_h}}}{\sin e + \frac{a_h}{\sin e + \frac{b_h}{\sin e + c_h}}}
\end{equation}

\begin{equation}
m_w(e) = \frac{1 + \frac{a_w}{1 + \frac{b_w}{1 + c_w}}}{\sin e + \frac{a_w}{\sin e + \frac{b_w}{\sin e + c_w}}}
\end{equation}
